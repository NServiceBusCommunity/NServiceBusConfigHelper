==== src/SampleEndpoint/Program.cs

// Generated by the NServiceBus wizard: https://nservicebuscommunity.github.io/NServiceBusConfigHelper/new?step=output&host=Worker&transport=Learning&persistence=NonDurable&serializer=SystemJson&tests=XunitV3&features=AttachmentsSql
using Microsoft.Data.SqlClient;
using NServiceBus.Attachments.Sql;
using SampleEndpoint;

var builder = Host.CreateApplicationBuilder(args);

var sqlServerConnectionString = ConnectionString("SqlServer");

var endpointConfiguration = new EndpointConfiguration("SampleEndpoint");

// Transport: Learning. https://docs.particular.net/transports/learning/
var transport = new LearningTransport();
endpointConfiguration.UseTransport(transport);

// Persistence: Non-durable. https://docs.particular.net/persistence/non-durable/
endpointConfiguration.UsePersistence<NonDurablePersistence>();

// Serializer: System.Text.Json. https://docs.particular.net/nservicebus/serialization/system-json
endpointConfiguration.UseSerialization<SystemJsonSerializer>();

// Where a message goes once every retry has failed. ServiceControl reads this queue.
endpointConfiguration.SendFailedMessagesTo("error");

// SQL attachments: https://github.com/NServiceBusCommunity/NServiceBus.Community.Attachments/blob/main/docs/sql.md
endpointConfiguration.EnableAttachments(
    () => new SqlConnection(sqlServerConnectionString),
    TimeToKeep.Default);

// Creates the queues, topics and tables the endpoint needs when it starts. Convenient in development.
// In production, run installers as a deployment step, with rights the running endpoint does not need.
endpointConfiguration.EnableInstallers();

builder.Services.AddNServiceBusEndpoint(endpointConfiguration);
builder.Services.AddSingleton<OrderSubmitter>();
builder.Services.AddHostedService<SampleOrders>();

await builder.Build().RunAsync();

string ConnectionString(string name)
{
    var value = builder.Configuration.GetConnectionString(name);
    if (string.IsNullOrWhiteSpace(value))
    {
        throw new InvalidOperationException(
            $"Connection string '{name}' is not set. Add it to appsettings.json, or set the ConnectionStrings__{name} environment variable.");
    }

    return value;
}

==== src/SampleEndpoint/Handlers/StoreDocumentHandler.cs

using SampleEndpoint.Messages;

namespace SampleEndpoint;

public class StoreDocumentHandler(ILogger<StoreDocumentHandler> logger) :
    IHandleMessages<StoreDocument>
{
    public async Task Handle(StoreDocument message, IMessageHandlerContext context)
    {
        // Read on demand: nothing is loaded until it is asked for, and a large attachment can
        // be copied to a stream instead with CopyTo.
        var attachments = context.Attachments();
        var content = await attachments.GetString("content", cancel: context.CancellationToken);
        logger.LogInformation("Document {Name} has {Length} characters", message.Name, content.Value.Length);
    }
}

==== src/SampleEndpoint/Attachments/DocumentSender.cs

using SampleEndpoint.Messages;

namespace SampleEndpoint;

/// <summary>Sends a <see cref="StoreDocument"/> with its content attached.</summary>
public static class DocumentSender
{
    public static Task Send(IMessageSession session, string name, string content)
    {
        var options = new SendOptions();
        options.RouteToThisEndpoint();
        var attachments = options.Attachments();
        attachments.AddString("content", content);
        return session.Send(
            new StoreDocument
            {
                Name = name
            },
            options);
    }
}
