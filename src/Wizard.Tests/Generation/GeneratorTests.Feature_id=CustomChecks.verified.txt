==== src/SampleEndpoint/Program.cs

// Generated by the NServiceBus wizard: https://nservicebuscommunity.github.io/NServiceBusConfigHelper/new?step=output&host=Worker&transport=Learning&persistence=NonDurable&serializer=SystemJson&tests=XunitV3&features=CustomChecks
using SampleEndpoint;

var builder = Host.CreateApplicationBuilder(args);

var endpointConfiguration = new EndpointConfiguration("SampleEndpoint");

// Transport: Learning. https://docs.particular.net/transports/learning/
var transport = new LearningTransport();
endpointConfiguration.UseTransport(transport);

// Persistence: Non-durable. https://docs.particular.net/persistence/non-durable/
endpointConfiguration.UsePersistence<NonDurablePersistence>();

// Serializer: System.Text.Json. https://docs.particular.net/nservicebus/serialization/system-json
endpointConfiguration.UseSerialization<SystemJsonSerializer>();

// Where a message goes once every retry has failed. ServiceControl reads this queue.
endpointConfiguration.SendFailedMessagesTo("error");

// Custom checks: https://docs.particular.net/monitoring/custom-checks/
endpointConfiguration.ReportCustomChecksTo("Particular.ServiceControl");

// Creates the queues, topics and tables the endpoint needs when it starts. Convenient in development.
// In production, run installers as a deployment step, with rights the running endpoint does not need.
endpointConfiguration.EnableInstallers();

builder.Services.AddNServiceBusEndpoint(endpointConfiguration);
builder.Services.AddSingleton<OrderSubmitter>();
builder.Services.AddHostedService<SampleOrders>();

await builder.Build().RunAsync();

==== src/SampleEndpoint/CustomChecks/DiskSpaceCheck.cs

using NServiceBus.CustomChecks;

namespace SampleEndpoint;

/// <summary>Runs every five minutes, and fails when the drive the endpoint runs from is nearly full.</summary>
public class DiskSpaceCheck() :
    CustomCheck("Disk space", "Infrastructure", TimeSpan.FromMinutes(5))
{
    const long minimumFreeBytes = 1_000_000_000;

    public override Task<CheckResult> PerformCheck(CancellationToken cancellationToken = default)
    {
        var drive = new DriveInfo(Path.GetPathRoot(AppContext.BaseDirectory)!);
        if (drive.AvailableFreeSpace < minimumFreeBytes)
        {
            return Task.FromResult(CheckResult.Failed($"Less than 1 GB free on {drive.Name}"));
        }

        return Task.FromResult(CheckResult.Pass);
    }
}


==== interactions

Warning service-control [Custom checks]
Custom checks sends to ServiceControl's queues. Until [ServiceControl](https://docs.particular.net/servicecontrol/) has been installed and has created them, those messages have nowhere to go.

Info development-only [Learning transport, Non-durable persistence]
The Learning transport and Non-durable persistence are for development. Choose a production one before deploying.
