==== src/SampleEndpoint/Program.cs

// Generated by the NServiceBus wizard: https://nservicebuscommunity.github.io/NServiceBusConfigHelper/new?step=output&host=Worker&transport=Learning&persistence=NonDurable&serializer=SystemJson&tests=XunitV3&features=Serilog
using SampleEndpoint;
using Serilog;

// Serilog: https://github.com/NServiceBusCommunity/NServiceBus.Community.Serilog
var loggerConfiguration = new LoggerConfiguration();
loggerConfiguration.Enrich.WithNsbExceptionDetails();
loggerConfiguration.WriteTo.Console();
Log.Logger = loggerConfiguration.CreateLogger();

var builder = Host.CreateApplicationBuilder(args);

// Serilog: https://github.com/NServiceBusCommunity/NServiceBus.Community.Serilog
// Serilog takes the place of the host's own providers, so each event is written once.
builder.Logging.ClearProviders();
builder.Logging.AddSerilog();

var endpointConfiguration = new EndpointConfiguration("SampleEndpoint");

// Transport: Learning. https://docs.particular.net/transports/learning/
var transport = new LearningTransport();
endpointConfiguration.UseTransport(transport);

// Persistence: Non-durable. https://docs.particular.net/persistence/non-durable/
endpointConfiguration.UsePersistence<NonDurablePersistence>();

// Serializer: System.Text.Json. https://docs.particular.net/nservicebus/serialization/system-json
endpointConfiguration.UseSerialization<SystemJsonSerializer>();

// Where a message goes once every retry has failed. ServiceControl reads this queue.
endpointConfiguration.SendFailedMessagesTo("error");

// Serilog: https://github.com/NServiceBusCommunity/NServiceBus.Community.Serilog
var serilogTracing = endpointConfiguration.EnableSerilogTracing(Log.Logger);
serilogTracing.EnableMessageTracing();

// Creates the queues, topics and tables the endpoint needs when it starts. Convenient in development.
// In production, run installers as a deployment step, with rights the running endpoint does not need.
endpointConfiguration.EnableInstallers();

builder.Services.AddNServiceBusEndpoint(endpointConfiguration);
builder.Services.AddSingleton<OrderSubmitter>();
builder.Services.AddHostedService<SampleOrders>();

await builder.Build().RunAsync();
