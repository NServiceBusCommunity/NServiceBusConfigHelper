==== src/SampleEndpoint/Program.cs

// Generated by the NServiceBus wizard: https://nservicebuscommunity.github.io/NServiceBusConfigHelper/new?step=output&host=Worker&transport=SqlServer&persistence=SqlServer&serializer=SystemJson&tests=XunitV3&features=Conventions,Sagas,HandlerOrdering,Recoverability,Outbox,Auditing,AuditFilter,Heartbeats,CustomChecks,Metrics,SagaAudit,OpenTelemetry,Serilog,FluentValidation,DataAnnotations,AttachmentsSql
using FluentValidation;
using HandlerOrdering;
using Microsoft.Data.SqlClient;
using NServiceBus.Attachments.Sql;
using NServiceBus.AuditFilter;
using OpenTelemetry.Metrics;
using OpenTelemetry.Resources;
using OpenTelemetry.Trace;
using SampleEndpoint;
using Serilog;

// Serilog: https://github.com/NServiceBusCommunity/NServiceBus.Community.Serilog
var loggerConfiguration = new LoggerConfiguration();
loggerConfiguration.Enrich.WithNsbExceptionDetails();
loggerConfiguration.WriteTo.Console();
Log.Logger = loggerConfiguration.CreateLogger();

var builder = Host.CreateApplicationBuilder(args);

var sqlServerConnectionString = ConnectionString("SqlServer");

// OpenTelemetry: https://docs.particular.net/nservicebus/operations/opentelemetry
builder.Services.AddOpenTelemetry()
    .ConfigureResource(resource => resource.AddService("SampleEndpoint"))
    .WithTracing(tracing => tracing
        .AddSource("NServiceBus.Core")
        .AddOtlpExporter())
    .WithMetrics(metering => metering
        .AddMeter("NServiceBus.Core.Pipeline.Incoming")
        .AddOtlpExporter());

// Serilog: https://github.com/NServiceBusCommunity/NServiceBus.Community.Serilog
// Serilog takes the place of the host's own providers, so each event is written once.
builder.Logging.ClearProviders();
builder.Logging.AddSerilog();

// FluentValidation: https://github.com/NServiceBusCommunity/NServiceBus.Community.Validation#nservicebusfluentvalidation
builder.Services.AddValidatorsFromAssemblyContaining<PlaceOrderValidator>();

var endpointConfiguration = new EndpointConfiguration("SampleEndpoint");

// Transport: SQL Server. https://docs.particular.net/transports/sql/
var transport = new SqlServerTransport(sqlServerConnectionString);
endpointConfiguration.UseTransport(transport);

// Persistence: SQL Server. https://docs.particular.net/persistence/sql/
var persistence = endpointConfiguration.UsePersistence<SqlPersistence>();
persistence.SqlDialect<SqlDialect.MsSqlServer>();
persistence.ConnectionBuilder(() => new SqlConnection(sqlServerConnectionString));

// Serializer: System.Text.Json. https://docs.particular.net/nservicebus/serialization/system-json
endpointConfiguration.UseSerialization<SystemJsonSerializer>();

// Where a message goes once every retry has failed. ServiceControl reads this queue.
endpointConfiguration.SendFailedMessagesTo("error");

// Message conventions: https://docs.particular.net/nservicebus/messaging/conventions
var conventions = endpointConfiguration.Conventions();
conventions.DefiningCommandsAs(type => type.Namespace == "SampleEndpoint.Messages.Commands");
conventions.DefiningEventsAs(type => type.Namespace == "SampleEndpoint.Messages.Events");

// Handler ordering: https://github.com/NServiceBusCommunity/NServiceBus.Community.HandlerOrdering
endpointConfiguration.ApplyInterfaceHandlerOrdering();

// Retry policy: https://docs.particular.net/nservicebus/recoverability/
var recoverability = endpointConfiguration.Recoverability();
recoverability.Immediate(immediate => immediate.NumberOfRetries(3));
recoverability.Delayed(delayed =>
{
    delayed.NumberOfRetries(2);
    delayed.TimeIncrease(TimeSpan.FromSeconds(15));
});

// Outbox: https://docs.particular.net/nservicebus/outbox/
// The outbox takes over from the transport's transaction, which then only covers the receive.
transport.TransportTransactionMode = TransportTransactionMode.ReceiveOnly;
endpointConfiguration.EnableOutbox();

// Auditing: https://docs.particular.net/nservicebus/operations/auditing
endpointConfiguration.AuditProcessedMessagesTo("audit");

// Audit filter: https://github.com/NServiceBusCommunity/NServiceBus.Community.AuditFilter
endpointConfiguration.FilterAuditQueue(FilterResult.IncludeInAudit);

// Heartbeats: https://docs.particular.net/monitoring/heartbeats/
endpointConfiguration.SendHeartbeatTo("Particular.ServiceControl");

// Custom checks: https://docs.particular.net/monitoring/custom-checks/
endpointConfiguration.ReportCustomChecksTo("Particular.ServiceControl");

// Performance metrics: https://docs.particular.net/monitoring/metrics/
var serviceControlMetrics = endpointConfiguration.EnableMetrics();
serviceControlMetrics.SendMetricDataToServiceControl("Particular.Monitoring", TimeSpan.FromSeconds(10));

// Saga audit: https://docs.particular.net/nservicebus/sagas/saga-audit
endpointConfiguration.AuditSagaStateChanges("Particular.ServiceControl");

// Serilog: https://github.com/NServiceBusCommunity/NServiceBus.Community.Serilog
var serilogTracing = endpointConfiguration.EnableSerilogTracing(Log.Logger);
serilogTracing.EnableMessageTracing();

// FluentValidation: https://github.com/NServiceBusCommunity/NServiceBus.Community.Validation#nservicebusfluentvalidation
endpointConfiguration.UseFluentValidation();

// DataAnnotations: https://github.com/NServiceBusCommunity/NServiceBus.Community.Validation#nservicebusdataannotations
endpointConfiguration.UseDataAnnotationsValidation();

// SQL attachments: https://github.com/NServiceBusCommunity/NServiceBus.Community.Attachments/blob/main/docs/sql.md
endpointConfiguration.EnableAttachments(
    () => new SqlConnection(sqlServerConnectionString),
    TimeToKeep.Default);

// Creates the queues, topics and tables the endpoint needs when it starts. Convenient in development.
// In production, run installers as a deployment step, with rights the running endpoint does not need.
endpointConfiguration.EnableInstallers();

builder.Services.AddNServiceBusEndpoint(endpointConfiguration);
builder.Services.AddSingleton<OrderSubmitter>();
builder.Services.AddHostedService<SampleOrders>();

await builder.Build().RunAsync();

string ConnectionString(string name)
{
    var value = builder.Configuration.GetConnectionString(name);
    if (string.IsNullOrWhiteSpace(value))
    {
        throw new InvalidOperationException(
            $"Connection string '{name}' is not set. Add it to appsettings.json, or set the ConnectionStrings__{name} environment variable.");
    }

    return value;
}

==== src/SampleEndpoint/OrderSubmitter.cs

using SampleEndpoint.Messages.Commands;

namespace SampleEndpoint;

/// <summary>
/// The one place that sends <see cref="PlaceOrder"/>. Code outside a handler sends through
/// <see cref="IMessageSession"/>; a handler uses the context it is given instead.
/// </summary>
public class OrderSubmitter(IMessageSession session)
{
    public async Task<Guid> Submit(string product, CancellationToken cancellationToken)
    {
        var command = new PlaceOrder
        {
            OrderId = Guid.NewGuid(),
            Product = product
        };
        // SendLocal puts the command on this endpoint's own queue.
        await session.SendLocal(command, cancellationToken);
        return command.OrderId;
    }
}

==== src/SampleEndpoint/Sagas/OrderSaga.cs

using SampleEndpoint.Messages.Events;

namespace SampleEndpoint;

/// <summary>
/// Starts when an order is placed, then waits for the cancellation window to pass. There is one
/// instance per order, found by <see cref="OrderSagaData.OrderId"/>.
/// </summary>
public class OrderSaga(ILogger<OrderSaga> logger) :
    Saga<OrderSagaData>,
    IAmStartedByMessages<OrderPlaced>,
    IHandleTimeouts<CancellationWindowElapsed>
{
    protected override void ConfigureHowToFindSaga(SagaPropertyMapper<OrderSagaData> mapper) =>
        mapper.MapSaga(saga => saga.OrderId)
            .ToMessage<OrderPlaced>(message => message.OrderId);

    public async Task Handle(OrderPlaced message, IMessageHandlerContext context)
    {
        logger.LogInformation("Order {OrderId}: waiting for the cancellation window", message.OrderId);
        await RequestTimeout<CancellationWindowElapsed>(context, TimeSpan.FromSeconds(10));
    }

    public Task Timeout(CancellationWindowElapsed state, IMessageHandlerContext context)
    {
        logger.LogInformation("Order {OrderId}: cancellation window elapsed, order confirmed", Data.OrderId);
        MarkAsComplete();
        return Task.CompletedTask;
    }
}

==== src/SampleEndpoint/CustomChecks/DiskSpaceCheck.cs

using NServiceBus.CustomChecks;

namespace SampleEndpoint;

/// <summary>Runs every five minutes, and fails when the drive the endpoint runs from is nearly full.</summary>
public class DiskSpaceCheck() :
    CustomCheck("Disk space", "Infrastructure", TimeSpan.FromMinutes(5))
{
    const long minimumFreeBytes = 1_000_000_000;

    public override Task<CheckResult> PerformCheck(CancellationToken cancellationToken = default)
    {
        var drive = new DriveInfo(Path.GetPathRoot(AppContext.BaseDirectory)!);
        if (drive.AvailableFreeSpace < minimumFreeBytes)
        {
            return Task.FromResult(CheckResult.Failed($"Less than 1 GB free on {drive.Name}"));
        }

        return Task.FromResult(CheckResult.Pass);
    }
}

using SampleEndpoint.Messages.Commands;

namespace SampleEndpoint;

public class StoreDocumentHandler(ILogger<StoreDocumentHandler> logger) :
    IHandleMessages<StoreDocument>
{
    public async Task Handle(StoreDocument message, IMessageHandlerContext context)
    {
        // Read on demand: nothing is loaded until it is asked for, and a large attachment can
        // be copied to a stream instead with CopyTo.
        var attachments = context.Attachments();
        var content = await attachments.GetString("content", cancel: context.CancellationToken);
        logger.LogInformation("Document {Name} has {Length} characters", message.Name, content.Value.Length);
    }
}

==== src/SampleEndpoint/Attachments/DocumentSender.cs

using SampleEndpoint.Messages.Commands;

namespace SampleEndpoint;

/// <summary>Sends a <see cref="StoreDocument"/> with its content attached.</summary>
public static class DocumentSender
{
    public static Task Send(IMessageSession session, string name, string content)
    {
        var options = new SendOptions();
        options.RouteToThisEndpoint();
        var attachments = options.Attachments();
        attachments.AddString("content", content);
        return session.Send(
            new StoreDocument
            {
                Name = name
            },
            options);
    }
}
