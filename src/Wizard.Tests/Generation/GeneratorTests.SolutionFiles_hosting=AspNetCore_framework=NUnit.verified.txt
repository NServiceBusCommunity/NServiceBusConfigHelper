==== src/SampleEndpoint/Program.cs

// Generated by the NServiceBus wizard: https://nservicebuscommunity.github.io/NServiceBusConfigHelper/new?step=output&host=AspNetCore&transport=Learning&persistence=Learning&serializer=SystemJson&tests=NUnit
using SampleEndpoint;

var builder = WebApplication.CreateBuilder(args);

var endpointConfiguration = new EndpointConfiguration("SampleEndpoint");

// Transport: Learning. https://docs.particular.net/transports/learning/
var transport = new LearningTransport();
endpointConfiguration.UseTransport(transport);

// Persistence: Learning. https://docs.particular.net/persistence/learning/
endpointConfiguration.UsePersistence<LearningPersistence>();

// Serializer: System.Text.Json. https://docs.particular.net/nservicebus/serialization/system-json
endpointConfiguration.UseSerialization<SystemJsonSerializer>();

// Where a message goes once every retry has failed. ServiceControl reads this queue.
endpointConfiguration.SendFailedMessagesTo("error");

// Creates the queues, topics and tables the endpoint needs when it starts. Convenient in development.
// In production, run installers as a deployment step, with rights the running endpoint does not need.
endpointConfiguration.EnableInstallers();

builder.Services.AddNServiceBusEndpoint(endpointConfiguration);
builder.Services.AddSingleton<OrderSubmitter>();

var app = builder.Build();

// POST /orders?product=Widget
app.MapPost(
    "/orders",
    async (string product, OrderSubmitter submitter, CancellationToken cancellationToken) =>
    {
        var orderId = await submitter.Submit(product, cancellationToken);
        return Results.Accepted(value: new {orderId});
    });

await app.RunAsync();

==== src/SampleEndpoint/OrderSubmitter.cs

using SampleEndpoint.Messages;

namespace SampleEndpoint;

/// <summary>
/// The one place that sends <see cref="PlaceOrder"/>. Code outside a handler sends through
/// <see cref="IMessageSession"/>; a handler uses the context it is given instead.
/// </summary>
public class OrderSubmitter(IMessageSession session)
{
    public async Task<Guid> Submit(string product, CancellationToken cancellationToken)
    {
        var command = new PlaceOrder
        {
            OrderId = Guid.NewGuid(),
            Product = product
        };
        // SendLocal puts the command on this endpoint's own queue.
        await session.SendLocal(command, cancellationToken);
        return command.OrderId;
    }
}

==== src/SampleEndpoint/Handlers/PlaceOrderHandler.cs

using SampleEndpoint.Messages;

namespace SampleEndpoint;

public class PlaceOrderHandler(ILogger<PlaceOrderHandler> logger) :
    IHandleMessages<PlaceOrder>
{
    public async Task Handle(PlaceOrder message, IMessageHandlerContext context)
    {
        logger.LogInformation("Order {OrderId}: placed for {Product}", message.OrderId, message.Product);

        // Publishing through the handler's context makes the event part of handling this message:
        // it is only sent once the handler has succeeded.
        await context.Publish(
            new OrderPlaced
            {
                OrderId = message.OrderId
            });
    }
}

==== src/SampleEndpoint.Tests/OrderSubmitterTests.cs

using NServiceBus.Testing;
using NUnit.Framework;
using SampleEndpoint.Messages;

namespace SampleEndpoint.Tests;

[TestFixture]
public class OrderSubmitterTests
{
    [Test]
    public async Task SendsPlaceOrder()
    {
        var session = new TestableMessageSession();
        var submitter = new OrderSubmitter(session);

        var orderId = await submitter.Submit("Widget", CancellationToken.None);

        var command = session.SentMessages.Single().Message<PlaceOrder>();
        Assert.That(command.OrderId, Is.EqualTo(orderId));
        Assert.That(command.Product, Is.EqualTo("Widget"));
    }
}

==== src/SampleEndpoint.Tests/PlaceOrderHandlerTests.cs

using Microsoft.Extensions.Logging.Abstractions;
using NServiceBus.Testing;
using NUnit.Framework;
using SampleEndpoint.Messages;

namespace SampleEndpoint.Tests;

[TestFixture]
public class PlaceOrderHandlerTests
{
    [Test]
    public async Task PublishesOrderPlaced()
    {
        var handler = new PlaceOrderHandler(NullLogger<PlaceOrderHandler>.Instance);
        var context = new TestableMessageHandlerContext();
        var command = new PlaceOrder
        {
            OrderId = Guid.NewGuid(),
            Product = "Widget"
        };

        await handler.Handle(command, context);

        var published = context.PublishedMessages.Single().Message<OrderPlaced>();
        Assert.That(published.OrderId, Is.EqualTo(command.OrderId));
    }
}
