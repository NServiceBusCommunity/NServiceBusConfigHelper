// Generated by the NServiceBus wizard: https://nservicebuscommunity.github.io/NServiceBusConfigHelper/new?step=output&host=Worker&transport=RabbitMQ&persistence=PostgreSql&serializer=SystemJson&tests=XunitV3&features=Sagas,Outbox,Serilog
using Npgsql;
using NpgsqlTypes;
using SampleEndpoint;
using Serilog;

// Serilog: https://github.com/NServiceBusCommunity/NServiceBus.Community.Serilog
var loggerConfiguration = new LoggerConfiguration();
loggerConfiguration.Enrich.WithNsbExceptionDetails();
loggerConfiguration.WriteTo.Console();
Log.Logger = loggerConfiguration.CreateLogger();

var builder = Host.CreateApplicationBuilder(args);

var rabbitMqConnectionString = ConnectionString("RabbitMQ");
var postgreSqlConnectionString = ConnectionString("PostgreSql");

// Serilog: https://github.com/NServiceBusCommunity/NServiceBus.Community.Serilog
// Serilog takes the place of the host's own providers, so each event is written once.
builder.Logging.ClearProviders();
builder.Logging.AddSerilog();

var endpointConfiguration = new EndpointConfiguration("SampleEndpoint");

// Transport: RabbitMQ. https://docs.particular.net/transports/rabbitmq/
var transport = new RabbitMQTransport(
    RoutingTopology.Conventional(QueueType.Quorum),
    rabbitMqConnectionString);
endpointConfiguration.UseTransport(transport);

// Persistence: PostgreSQL. https://docs.particular.net/persistence/sql/
var persistence = endpointConfiguration.UsePersistence<SqlPersistence>();
var dialect = persistence.SqlDialect<SqlDialect.PostgreSql>();
// Saga data is stored as jsonb, which Npgsql has to be told for each parameter.
dialect.JsonBParameterModifier(parameter =>
{
    var npgsqlParameter = (NpgsqlParameter) parameter;
    npgsqlParameter.NpgsqlDbType = NpgsqlDbType.Jsonb;
});
persistence.ConnectionBuilder(() => new NpgsqlConnection(postgreSqlConnectionString));

// Serializer: System.Text.Json. https://docs.particular.net/nservicebus/serialization/system-json
endpointConfiguration.UseSerialization<SystemJsonSerializer>();

// Where a message goes once every retry has failed. ServiceControl reads this queue.
endpointConfiguration.SendFailedMessagesTo("error");

// Outbox: https://docs.particular.net/nservicebus/outbox/
// The outbox takes over from the transport's transaction, which then only covers the receive.
transport.TransportTransactionMode = TransportTransactionMode.ReceiveOnly;
endpointConfiguration.EnableOutbox();

// Serilog: https://github.com/NServiceBusCommunity/NServiceBus.Community.Serilog
var serilogTracing = endpointConfiguration.EnableSerilogTracing(Log.Logger);
serilogTracing.EnableMessageTracing();

// Creates the queues, topics and tables the endpoint needs when it starts. Convenient in development.
// In production, run installers as a deployment step, with rights the running endpoint does not need.
endpointConfiguration.EnableInstallers();

builder.Services.AddNServiceBusEndpoint(endpointConfiguration);
builder.Services.AddSingleton<OrderSubmitter>();
builder.Services.AddHostedService<SampleOrders>();

await builder.Build().RunAsync();

string ConnectionString(string name)
{
    var value = builder.Configuration.GetConnectionString(name);
    if (string.IsNullOrWhiteSpace(value))
    {
        throw new InvalidOperationException(
            $"Connection string '{name}' is not set. Add it to appsettings.json, or set the ConnectionStrings__{name} environment variable.");
    }

    return value;
}
